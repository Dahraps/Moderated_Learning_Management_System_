cmake_minimum_required(VERSION 3.16)
project(MLMS VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED)
add_library(mlms_core src/LMS.cpp src/Console.cpp)
target_include_directories(mlms_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(mlms_core PUBLIC SQLite::SQLite3 OpenSSL::Crypto)
if(MSVC)
 target_compile_options(mlms_core PRIVATE /W4)
else()
 target_compile_options(mlms_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(mlms src/main.cpp)
target_link_libraries(mlms PRIVATE mlms_core)
enable_testing()
add_executable(mlms_tests tests/test_logic.cpp)
target_link_libraries(mlms_tests PRIVATE mlms_core)
add_test(NAME mlms_tests COMMAND mlms_tests)

add_executable(mlms_role_tests tests/test_roles.cpp)
target_link_libraries(mlms_role_tests PRIVATE mlms_core)
add_test(NAME mlms_role_tests COMMAND mlms_role_tests)
set_tests_properties(mlms_role_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(mlms_srs_tests tests/test_srs.cpp)
target_link_libraries(mlms_srs_tests PRIVATE mlms_core)
add_test(NAME mlms_srs_tests COMMAND mlms_srs_tests)
set_tests_properties(mlms_srs_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(mlms_rule_tests tests/test_rules.cpp)
target_link_libraries(mlms_rule_tests PRIVATE mlms_core)
add_test(NAME mlms_rule_tests COMMAND mlms_rule_tests)
set_tests_properties(mlms_rule_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
